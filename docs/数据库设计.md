## 一、实体关系图（ER Diagram）

### 核心业务关系说明

**出口合同与门店的关系（重要变更）：**
- 一个出口合同（SalesContract）可以包含多个门店的商品
- 门店信息记录在销售明细（SalesItem）中，而非合同表
- 这样设计可以支持一个合同服务多个门店的场景

**示例场景：**
```
出口合同 EXP2500001（混合发货）
├── 禧瑞都门店
│   ├── 瓷砖800*800，100箱
│   └── 洗手盆，50箱
└── 安纳汉姆门店
    ├── 马桶，30箱
    └── 浴室柜，20箱
```

## 二、Prisma Schema

```prisma
// prisma/schema.prisma

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "sqlite"
  url      = env("DATABASE_URL")
}

// ==================== 用户与权限 ====================

model User {
  id            String    @id @default(cuid())
  username      String    @unique
  password      String    // bcrypt加密
  name          String    // 显示名称
  role          Role      @default(SALES)
  email         String?
  phone         String?
  avatar        String?
  isActive      Boolean   @default(true)
  lastLoginAt   DateTime?
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt

  // 关联
  operationLogs OperationLog[]
  chatHistories ChatHistory[]
  notifications Notification[]

  @@map("users")
}

enum Role {
  ADMIN     // 管理员/老板
  PURCHASE  // 采购
  SALES     // 销售
}

// ==================== 基础数据 ====================

model Port {
  id        String   @id @default(cuid())
  name      String   @unique  // 港口名称：洛杉矶、Oakland、密歇根
  code      String   @unique  // 简码：LA、OAK、MI
  isActive  Boolean  @default(true)
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  // 关联
  stores     Store[]
  containers Container[]

  @@map("ports")
}

model Supplier {
  id              String    @id @default(cuid())
  name            String    // 正式名称（公司名）
  shortName       String?   // 简称
  contactName     String?   // 联系人
  contactPhone    String?   // 联系电话
  contactEmail    String?   // 联系邮箱
  address         String?   // 地址
  bankAccount     String?   // 银行账户
  hasQualityIssue Boolean   @default(false)  // 质量问题标记
  qualityNote     String?   // 质量问题备注
  isActive        Boolean   @default(true)
  createdAt       DateTime  @default(now())
  updatedAt       DateTime  @updatedAt

  // 关联
  aliases           SupplierAlias[]
  purchaseContracts PurchaseContract[]
  products          ProductSupplier[]

  @@map("suppliers")
}

model SupplierAlias {
  id         String   @id @default(cuid())
  alias      String   @unique  // 昵称：黎总、叶总等
  supplierId String
  createdAt  DateTime @default(now())

  // 关联
  supplier Supplier @relation(fields: [supplierId], references: [id])

  @@map("supplier_aliases")
}

model Store {
  id           String   @id @default(cuid())
  name         String   @unique  // 门店名称
  portId       String   // 所属港口
  contactName  String?  // 联系人
  contactPhone String?  // 联系电话
  contactEmail String?  // 联系邮箱
  address      String?  // 地址
  isActive     Boolean  @default(true)
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  // 关联
  port      Port        @relation(fields: [portId], references: [id])
  salesItems SalesItem[]  // 通过销售明细关联出口合同

  @@map("stores")
}

model Product {
  id             String   @id @default(cuid())
  customsName    String   // 报关名
  description    String?  // 商品补充信息
  specification  String?  // 规格
  unit           String?  // 单位
  categoryId     String?  // 分类ID
  isActive       Boolean  @default(true)
  createdAt      DateTime @default(now())
  updatedAt      DateTime @updatedAt

  // 关联
  category        ProductCategory?  @relation(fields: [categoryId], references: [id])
  suppliers       ProductSupplier[]
  purchaseItems   PurchaseItem[]
  salesItems      SalesItem[]
  inventories     Inventory[]
  containerItems  ContainerItem[]
  priceHistories  PriceHistory[]

  @@map("products")
}

model ProductCategory {
  id        String   @id @default(cuid())
  name      String   @unique
  parentId  String?
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  // 关联
  parent   ProductCategory?  @relation("CategoryHierarchy", fields: [parentId], references: [id])
  children ProductCategory[] @relation("CategoryHierarchy")
  products Product[]

  @@map("product_categories")
}

// 商品-供应商关联表（多对多，含报价信息）
model ProductSupplier {
  id         String   @id @default(cuid())
  productId  String
  supplierId String
  price      Float?   // 最新采购价格
  updatedAt  DateTime @updatedAt

  // 关联
  product  Product  @relation(fields: [productId], references: [id])
  supplier Supplier @relation(fields: [supplierId], references: [id])

  @@unique([productId, supplierId])
  @@map("product_suppliers")
}

// 采购价格历史
model PriceHistory {
  id         String   @id @default(cuid())
  productId  String
  supplierId String?
  price      Float
  recordedAt DateTime @default(now())

  // 关联
  product Product @relation(fields: [productId], references: [id])

  @@map("price_histories")
}

// ==================== 采购管理 ====================

model PurchaseContract {
  id             String              @id @default(cuid())
  contractNo     String              @unique  // 合同编号：CG25XXXXX
  supplierId     String
  totalAmount    Float               @default(0)  // 合同总金额
  paidAmount     Float               @default(0)  // 已付金额
  status         PurchaseStatus      @default(DRAFT)
  signedAt       DateTime?           // 签订日期
  expectedDate   DateTime?           // 预计交货日期
  invoiceNo      String?             // 发票号码
  note           String?             // 备注
  createdAt      DateTime            @default(now())
  updatedAt      DateTime            @updatedAt

  // 关联
  supplier   Supplier         @relation(fields: [supplierId], references: [id])
  items      PurchaseItem[]
  payments   Payment[]
  files      ContractFile[]

  @@map("purchase_contracts")
}

enum PurchaseStatus {
  DRAFT       // 草稿
  SIGNED      // 已签订
  PRODUCING   // 生产中
  SHIPPED     // 已发货
  RECEIVED    // 已收货
  COMPLETED   // 已完成
  CANCELLED   // 已取消
}

model PurchaseItem {
  id                 String   @id @default(cuid())
  purchaseContractId String
  productId          String
  quantity           Float    // 数量
  unit               String?  // 单位
  unitPrice          Float    // 单价
  totalPrice         Float    // 总价
  specification      String?  // 规格
  note               String?  // 备注
  createdAt          DateTime @default(now())
  updatedAt          DateTime @updatedAt

  // 关联
  purchaseContract PurchaseContract @relation(fields: [purchaseContractId], references: [id], onDelete: Cascade)
  product          Product          @relation(fields: [productId], references: [id])
  inventories      Inventory[]

  @@map("purchase_items")
}

// ==================== 销售管理 ====================

model SalesContract {
  id             String         @id @default(cuid())
  contractNo     String         @unique  // 合同编号：EXP25XXXXX
  totalAmount    Float          @default(0)  // 合同总金额（美元）
  receivedAmount Float          @default(0)  // 已收金额
  exchangeRate   Float          // 使用的汇率
  status         SalesStatus    @default(DRAFT)
  signedAt       DateTime?      // 签订日期
  note           String?        // 备注
  createdAt      DateTime       @default(now())
  updatedAt      DateTime       @updatedAt

  // 关联
  items    SalesItem[]
  payments Payment[]

  @@map("sales_contracts")
}

enum SalesStatus {
  DRAFT       // 草稿
  CONFIRMED   // 已确认
  PAID        // 已收款
  SHIPPED     // 已发货
  COMPLETED   // 已完成
  CANCELLED   // 已取消
}

model SalesItem {
  id              String   @id @default(cuid())
  salesContractId String
  productId       String
  storeId         String   // 门店ID（一个出口合同可包含多个门店的商品）
  quantity        Float    // 数量
  unit            String?  // 单位
  costPrice       Float    // 成本价（人民币）
  sellingPrice    Float    // 销售价（美元）
  specification   String?  // 规格
  note            String?  // 备注
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  // 关联
  salesContract SalesContract @relation(fields: [salesContractId], references: [id], onDelete: Cascade)
  product       Product       @relation(fields: [productId], references: [id])
  store         Store         @relation(fields: [storeId], references: [id])
  inventories   Inventory[]

  @@map("sales_items")
}

// ==================== 库存管理 ====================

model Inventory {
  id               String          @id @default(cuid())
  productId        String
  purchaseItemId   String?         // 关联采购明细
  salesItemId      String?         // 关联销售明细（出库时）
  containerId      String?         // 关联货柜（出库时）
  quantity         Float           // 数量
  unit             String?         // 单位
  status           InventoryStatus @default(PRODUCING)
  inboundAt        DateTime?       // 入库时间
  outboundAt       DateTime?       // 出库时间
  note             String?         // 备注
  createdAt        DateTime        @default(now())
  updatedAt        DateTime        @updatedAt

  // 关联
  product      Product       @relation(fields: [productId], references: [id])
  purchaseItem PurchaseItem? @relation(fields: [purchaseItemId], references: [id])
  salesItem    SalesItem?    @relation(fields: [salesItemId], references: [id])
  container    Container?    @relation(fields: [containerId], references: [id])

  @@map("inventories")
}

enum InventoryStatus {
  PRODUCING   // 生产中
  PACKING     // 包装中
  SHIPPING    // 运输中
  INBOUND     // 已入库
  OUTBOUND    // 已出库
}

// ==================== 货柜管理 ====================

model Container {
  id              String          @id @default(cuid())
  containerNo     String          @unique  // 货柜编号：25-001-LA
  portId          String
  status          ContainerStatus @default(PENDING)
  totalBoxes      Int             @default(0)   // 总箱数
  grossWeight     Float           @default(0)   // 毛重
  netWeight       Float           @default(0)   // 净重
  volume          Float           @default(0)   // 体积
  shippedAt       DateTime?       // 发运日期
  estimatedArrival DateTime?      // 预计到达时间
  note            String?         // 备注
  createdAt       DateTime        @default(now())
  updatedAt       DateTime        @updatedAt

  // 关联
  port       Port            @relation(fields: [portId], references: [id])
  items      ContainerItem[]
  inventories Inventory[]

  // 报关信息（嵌入）
  customsBroker   String?   // 报关公司
  isFumigated     Boolean?  // 是否熏蒸
  hasTaxRefund    Boolean?  // 是否报出口退税

  @@map("containers")
}

enum ContainerStatus {
  PENDING   // 待装柜
  LOADING   // 装柜中
  SHIPPED   // 已发运
  ARRIVED   // 已到达
}

model ContainerItem {
  id          String   @id @default(cuid())
  containerId String
  productId   String
  storeId     String?  // 门店（一柜可装多店）
  quantity    Float    // 数量
  unit        String?  // 单位
  boxes       Int?     // 箱数
  grossWeight Float?   // 毛重
  netWeight   Float?   // 净重
  volume      Float?   // 体积
  note        String?  // 备注
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  // 关联
  container Container @relation(fields: [containerId], references: [id], onDelete: Cascade)
  product   Product   @relation(fields: [productId], references: [id])

  @@map("container_items")
}

// ==================== 财务管理 ====================

model Payment {
  id                 String       @id @default(cuid())
  type               PaymentType  // 付款类型
  purchaseContractId String?      // 关联采购合同
  salesContractId    String?      // 关联销售合同
  amount             Float        // 金额
  currency           String       @default("CNY")  // 币种
  paymentMethod      String?      // 付款方式
  paymentDate        DateTime     // 付款日期
  note               String?      // 备注（如：定金、尾款）
  createdAt          DateTime     @default(now())
  updatedAt          DateTime     @updatedAt

  // 关联
  purchaseContract PurchaseContract? @relation(fields: [purchaseContractId], references: [id])
  salesContract    SalesContract?    @relation(fields: [salesContractId], references: [id])

  @@map("payments")
}

enum PaymentType {
  PAYABLE    // 应付（付给供应商）
  RECEIVABLE // 应收（从门店收款）
}

// ==================== 文件管理 ====================

model ContractFile {
  id                 String   @id @default(cuid())
  purchaseContractId String
  fileName           String   // 原始文件名
  filePath           String   // 存储路径
  fileType           String   // 文件类型
  fileSize           Int      // 文件大小（字节）
  uploadedAt         DateTime @default(now())

  // 关联
  purchaseContract PurchaseContract @relation(fields: [purchaseContractId], references: [id], onDelete: Cascade)

  @@map("contract_files")
}

// ==================== AI与日志 ====================

model ChatHistory {
  id        String   @id @default(cuid())

  action     String   // 操作类型
  entity     String   // 操作实体

  newValue   String?  // 新值（JSON）

  userAgent  String?  // 用户代理
  createdAt  DateTime @default(now())

  user User @relation(fields: [userId], references: [id])

  @@index([entity, entityId])

// ==================== 系统配置 ====================

model SystemConfig {
  id        String   @id @default(cuid())

// ==================== 通知 ====================
```

## 三、核心表说明

### 3.1 用户表（users）

| 字段 | 类型 | 说明 |
|------|------|------|
| id | String | 主键 |
| username | String | 用户名（唯一） |
| password | String | 密码（bcrypt加密） |
| name | String | 显示名称 |
| role | Enum | 角色：ADMIN/PURCHASE/SALES |
| email | String? | 邮箱 |
| isActive | Boolean | 是否启用 |

### 3.2 供应商表（suppliers）

| 字段 | 类型 | 说明 |
|------|------|------|
| id | String | 主键 |
| name | String | 正式名称（公司名） |
| shortName | String? | 简称 |
| hasQualityIssue | Boolean | 质量问题标记 |
| qualityNote | String? | 质量问题备注 |

### 3.3 采购合同表（purchase_contracts）

| 字段 | 类型 | 说明 |
|------|------|------|
| id | String | 主键 |
| contractNo | String | 合同编号（CG25XXXXX） |
| supplierId | String | 供应商ID |
| totalAmount | Float | 合同总金额 |
| paidAmount | Float | 已付金额 |
| status | Enum | 状态 |

### 3.4 出口合同表（sales_contracts）

> 说明：一个出口合同可包含多个门店，门店信息记录在 `sales_items.storeId` 中。

| 字段 | 类型 | 说明 |
|------|------|------|
| id | String | 主键 |

## 六、初始数据（Seed）

```typescript
// prisma/seed.ts

import { PrismaClient } from '@prisma/client';
import bcrypt from 'bcrypt';

const prisma = new PrismaClient();

async function main() {
  // 1. 创建管理员用户
  const adminPassword = await bcrypt.hash('admin123', 12);
  await prisma.user.upsert({
    where: { username: 'admin' },
    update: {},
    create: {
      username: 'admin',
      password: adminPassword,
      name: '管理员',
      role: 'ADMIN',
    },
  });

  // 2. 创建港口
  const ports = [
    { name: '洛杉矶', code: 'LA' },
    { name: 'Oakland', code: 'OAK' },
    { name: '密歇根', code: 'MI' },
  ];
  
  for (const port of ports) {
    await prisma.port.upsert({
      where: { code: port.code },
      update: {},
      create: port,
    });
  }

  // 3. 创建系统配置
  const configs = [
    { key: 'exchangeRate', value: JSON.stringify({ rate: 6.8, auto: true }), note: '汇率配置' },
    { key: 'profitRate', value: JSON.stringify({ rate: 1.3 }), note: '利润率配置' },
    { key: 'contractPrefix', value: JSON.stringify({ purchase: 'CG', sales: 'EXP' }), note: '合同编号前缀' },
  ];
  
  for (const config of configs) {
    await prisma.systemConfig.upsert({
      where: { key: config.key },
      update: { value: config.value },
      create: config,
    });
  }

  console.log('Seed completed');
}

main()
  .catch((e) => {
    console.error(e);
    process.exit(1);
  })
  .finally(async () => {
    await prisma.$disconnect();
  });
```
